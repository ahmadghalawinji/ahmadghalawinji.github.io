Code Sample

Cohort Retention Query (SQL)

SQL · BigQuery
-- Build weekly cohorts and measure retention
WITH cohorts AS (
  SELECT
    user_id,
    DATE_TRUNC(first_visit, WEEK) AS cohort_week
  FROM (
    SELECT user_id, MIN(session_date) AS first_visit
    FROM `analytics.sessions`
    GROUP BY 1
  )
),
activity AS (
  SELECT
    s.user_id,
    c.cohort_week,
    DATE_DIFF(DATE_TRUNC(s.session_date, WEEK), c.cohort_week, WEEK) AS weeks_since_join
  FROM `analytics.sessions` s
  JOIN cohorts c USING (user_id)
)
SELECT
  cohort_week,
  weeks_since_join,
  COUNT(DISTINCT user_id) AS active_users
FROM activity
WHERE weeks_since_join BETWEEN 0 AND 12
GROUP BY 1, 2
ORDER BY 1, 2
